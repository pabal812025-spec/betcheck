function livePressure(){ let chances = (homeChances-awayChances)*0.65; let shots = (homeShots-awayShots)*0.25; let target = (homeTarget-awayTarget)*1.10; let corners = (homeCorners-awayCorners)*0.25; let possession = (homePoss-awayPoss)*0.06; /* Las expulsiones tienen bastante peso, pero nunca llevan automáticamente una probabilidad a 0%. */ let redCards = (awayReds-homeReds)*14; return chances + shots + target + corners + possession + redCards; }
