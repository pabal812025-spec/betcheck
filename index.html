// ============================================================ // TRAMO FINAL DEL PARTIDO // Solo afecta desde 85' en adelante. // NO modifica la calibración anterior. // ============================================================ if (testState.status === "live" && minute >= 85) { const remainingFactor = Math.max(0, Math.min(1, (minute - 85) / 10)); // Cuanto menos tiempo queda, más peso tiene el resultado actual. // 85' -> 0.00 // 87' -> 0.20 // 89' -> 0.40 // 90' -> 0.50 // 95' -> 1.00 // // Después de 90 se interpreta como tiempo añadido. if (hg === ag) { // Empate: al final del partido el empate debe dominar claramente. // 85' -> ~82% // 89' -> ~96% // 90' -> ~97% // 95' -> ~98% const lateDraw = 82 + Math.min(16, remainingFactor * 32); const nonDraw = Math.max(0.001, home + away); const remaining = 100 - lateDraw; home = (home / nonDraw) * remaining; away = (away / nonDraw) * remaining; draw = lateDraw; } else { // Hay un ganador. // El equipo que va ganando debe aumentar mucho su probabilidad // conforme desaparece el tiempo disponible. if (hg > ag) { const winnerTarget = 82 + Math.min(16, remainingFactor * 32); const others = Math.max(0.001, draw + away); const remaining = 100 - winnerTarget; draw = (draw / others) * remaining; away = (away / others) * remaining; home = winnerTarget; } else { const winnerTarget = 82 + Math.min(16, remainingFactor * 32); const others = Math.max(0.001, home + draw); const remaining = 100 - winnerTarget; home = (home / others) * remaining; draw = (draw / others) * remaining; away = winnerTarget; } } } // ============================================================ // PARTIDO FINALIZADO // Solo FINALIZADO puede dar 100%. // ============================================================ if (testState.status === "finished") { if (hg > ag) { home = 100; draw = 0; away = 0; } else if (ag > hg) { home = 0; draw = 0; away = 100; } else { home = 0; draw = 100; away = 0; } }
